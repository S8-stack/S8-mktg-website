// LaunchServer.java
import com.s8.meta.api.S8;
import com.s8.meta.api.server.S8BootMethod;
import com.s8.meta.api.server.S8WebServer;

public class LaunchServer {
  public static void main(String[] args) throws Exception {
    S8WebServer server = S8.WEB_SERVER_FACTORY.createWebServer(
        "/path/to/app-config.xml",
        LaunchServer.class,
        params -> params.addBootMethod(
            S8BootMethod.MAIN_BOOT_PAGE, AppBoot.INSTANCE));
    server.start();
  }
}

// AppBoot.java — the front is a remote screen
import com.s8.meta.api.annotations.S8ObjectType;
import com.s8.meta.api.flow.S8SyncFlow;
import com.s8.meta.api.front.S8FrontObject;
import com.s8.meta.api.front.S8FrontSession;
import com.s8.meta.api.server.S8BootMethod;

public final class AppBoot implements S8BootMethod {
  static final AppBoot INSTANCE = new AppBoot();

  @Override
  public void boot(S8SyncFlow flow) throws Exception {
    S8FrontSession front = flow.front();
    HomeScreen screen = new HomeScreen(front);
    screen.setTitle("Hello S8");
    screen.vertex.expose(0);
    flow.refresh();
  }
}

@S8ObjectType(name = "app.HomeScreen")
class HomeScreen extends S8FrontObject {
  HomeScreen(S8FrontSession front) {
    super(front, "app.HomeScreen");
  }
  void setTitle(String title) {
    vertex.outbound().setStringUTF8Field("title", title);
  }
}

// Flash persistency — annotate a row, put it, get it
@S8RowType(name = "user-v1")
public class UserRow extends RowS8Object {
  public String email;
  public UserRow(String id) { super(id); }
}

TablesDB db = flow.tables();
db.createTable(new CreateTableS8Request("users", true));

UserRow row = new UserRow("u-42");
row.email = "ada@stack8.tech";
db.putRow(new PutRowS8Request("users", row, true, true));

GetRowS8Response got = db.getRow("users", "u-42");

// Spaces hold object graphs — logical, stored, versioned
SpacesDB spaces = flow.spaces();
spaces.createSpace(new CreateSpaceS8Request(
    "space/ada", new SpaceS8Object[]{ root }));

AccessSpaceS8Response res =
    spaces.accessSpace(new AccessSpaceS8Request("space/ada", true));

if (res.status == AccessSpaceS8Response.Status.OK) {
  Object[] exposed = res.objects;
}

flow.refresh();

A stack, not a kit

S8 abstracts the parts every serious web product rediscovers: HTTP/2 and TLS, request mapping, the thread model, databases, and the front/back API. You write against one flow.

Simple

Less is More

JAVA Core to code the entire app (back, front, db). A unified API to code and deploy. Launched in no time

Secure

Paranoid mode enabled

Zero dependencies means bullet-proof defense against vicious open source vulnerabilities insertion in foundations libraries. And the basics: encryption...

Small

Small-Surface API

The number of things you need to know to start coding on the S8 Stack is astonishingly low! Even if you start a top tier complexity project. So you can build features instead of doing IT stuff

Synced

High concurrency architecture

Because scaling to high volume requires to think asynchronous since day 1, we have done it for you under the hood (you don't even have to know what a light thread is).

Storage

Embedded DB and caching

Annotated your objects and we provide under the hood the most efficient caching and databases accesses.

Structured

Graph DB, native version

Handle large objects in db with arbitrary graph shapes. Attach resources directly to objects, leveraging S8 full binary architecture. Go back in time on natively versioned data.

Screen

Your front as a remote screen

Handle directly your front components from the back. Use standards to create new ones with extension mechanisms.

Scalable

Lake of nodes

Use cluster of S8 instances to achieve resilience, load balancing and back-up at arbitrary level.
